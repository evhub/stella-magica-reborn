#Magi Organization Center
stella_building_magi_center_1 = {
	base_buildtime = 240
	#base_cap_amount = 1
	#icon = gfx_stella_building_magi_center_1
	category = government

	# PORT (4.6): every building needs a building set (decides which slots/zones accept it)
	building_sets = {
		government
	}
	
	potential = {
		exists = owner
		NOT = { has_modifier = slave_colony }
		owner = { has_valid_civic = civic_magical_legacy }
	}

	allow = {
		has_upgraded_capital = yes
		#For some reason doing base_cap_amount prevents upgrading buildings, so I have to use this janky thing instead
		NOR = {
			has_building = stella_building_magi_center_1
			has_building = stella_building_magi_center_2
			has_building = stella_building_magi_center_3
		}
	}
	
	planet_modifier = {
	
			job_stella_magi_coord_add = 100
	}
	
	resources = {
		category = planet_buildings
		cost = {
			minerals = 400
		}
		upkeep = {
			energy = 3
			consumer_goods = 2
		}
	}
	
	prerequisites = {
		stella_tech_magi_organization
	}	
	
	triggered_desc = {
		trigger = { 
			exists = owner
			owner = { is_regular_empire = yes } 
		}
		text = job_stella_magi_coord_effect_desc
	}

	ai_weight = {
		weight = 1
		modifier = 
		{
			factor = 6
			owner = { 
				has_monthly_income = {
					resource = stella_grief_cube
					value < 6
				}
			}
		}
	}

	ai_resource_production = {
		stella_grief_cube = 1
		trigger = {
			always = yes
		}
	}
	
	show_tech_unlock_if = { has_valid_civic = civic_magical_legacy }
	
	upgrades = {
		stella_building_magi_center_2
	}
}
