# Grief Cube Rationing: strict allotments for the armed forces stretch the stockpile, at some cost to morale.
tts_grief_cube_rationing = {
	length = 3600
	icon = "GFX_edict_type_time"
	resources = {
		category = edicts
		cost = {
			influence = 100
		}
	}

	modifier = {
		armies_grief_cube_upkeep_mult = -0.25	# MagOps squads are the main grief cube upkeep
		pop_happiness = -0.03
	}

	potential = {
		has_valid_civic = civic_magical_legacy
	}

	show_tech_unlock_if = { has_valid_civic = civic_magical_legacy }

	prerequisites = {
		"stella_tech_grief_studies"
	}

	ai_weight = {
		weight = 0
		modifier = {
			weight = 1
			has_resource = { type = grief_cube amount < 50 }
		}
	}
}
