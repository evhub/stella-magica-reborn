# Governance: humanity's AI-advised representative government. Elected human
# representatives and advisory intelligences deliberate together and publish their reasoning.
civic_tts_governance = {
	icon = "gfx/interface/icons/governments/civics/civic_technocracy.dds"
	description = civic_tooltip_tts_governance_effects

	potential = {
		ethics = { NOT = { value = ethic_gestalt_consciousness } }
		authority = { NOT = { value = auth_corporate } }
		# The Unified Earth origin already includes Governance.
		origin = { NOT = { value = origin_tts_unified_earth } }
	}
	possible = {
		authority = {
			OR = {
				value = auth_democratic
				value = auth_oligarchic
			}
		}
	}

	random_weight = { base = 5 }

	modifier = {
		country_admin_cap_add = 10
		planet_stability_add = 3
		country_edict_fund_add = 50
	}
}

# Mahou Shoujo Youkai (MSY): the magical girls' mutual aid society, which shares grief cubes so

	description = civic_tooltip_tts_msy_effects

	potential = {
		ethics = { NOT = { value = ethic_gestalt_consciousness } }
	}
	possible = {
		civics = {
			value = civic_magical_legacy
		}
	}

	random_weight = { base = 0 }

	modifier = {
		country_grief_cube_produces_mult = 0.20
		army_morale = 0.10
	}
}


# Origin: The Unified Earth. Humanity ended its wars by building something better: one
# government, advised by its own creations, answerable to everyone.
origin_tts_unified_earth = {
	is_origin = yes
	icon = "gfx/interface/icons/origins/origins_golden_generation.dds"
	picture = GFX_evt_metropolis
	description = origin_tooltip_tts_unified_earth_effects

	potential = {
		always = yes
	}
	possible = {
		ethics = { NOT = { value = ethic_gestalt_consciousness } }
	}

	random_weight = { base = 2 }

	# Governance grew out of the end of the Unification Wars: this origin includes it,
	# with the same effects as the Governance civic, without using a civic slot.
	modifier = {
		country_unity_produces_mult = 0.05
		army_damage_mult = 0.10
		country_admin_cap_add = 10
		planet_stability_add = 3
		country_edict_fund_add = 50
	}
}
