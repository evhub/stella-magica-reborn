	}
}

# This = owner of fleet 1 (combatant)
# From = owner of fleet 2 (destroyed)
# FromFrom = fleet 1
# FromFromFrom = fleet 2
on_fleet_destroyed_perp = {
	events = {
		stella_story.11  #Drones defeated
		stella_story.15  #Crystals defeated
		stella_dim.7     #Void Clouds defeated
		stella_dim.18    #Pisonic Avatar defeated
		stella_dim.23	 #Horror destroyed
		stella_apop.6    #Invaders destroyed
		stella_crisis.12 #Portal destroyed
		stella_crisis.13 #Portal destroyed (by others)
	}
	
}

# Triggers when trying to open the diplomacy view for countries with custom diplomacy.
# Country scope
# This = target country (player)
# From = source country

	events = {
		stella_leader.1000 #Madoka Diplomacy
	}
}
