#Incubators get a slight opinion boost, more towards materialists, but spiritualist empires think they are evil

stella_wardens_opinion = {
	trigger = {
		is_country_type = default
		NOR = {
			has_ethic = "ethic_spiritualist"
			has_ethic = "ethic_fanatic_spiritualist"
		}
		FROM = { has_civic = civic_hive_entropic_wardens }
	}
	opinion = {
		base = 10
		
		modifier = {
			add = 10
			has_ethic = "ethic_materialist"
			FROM = { has_civic = civic_hive_entropic_wardens }
		}
		
		modifier = {
			add = 20
			has_ethic = "ethic_fanatic_materialist"
			FROM = { has_civic = civic_hive_entropic_wardens }
		}
		
		
	}

}

stella_wardens_opinion_spirtualist = {
	trigger = {
		is_country_type = default
		FROM = { has_civic = civic_hive_entropic_wardens }
	}
	
	opinion = {
	base = 0
		modifier = {
			add = -10
			has_ethic = "ethic_spiritualist"
			FROM = { has_civic = civic_hive_entropic_wardens }
		}
		
		modifier = {
			add = -20
			has_ethic = "ethic_fanatic_spiritualist"
			FROM = { has_civic = civic_hive_entropic_wardens }
		}
	}
}
