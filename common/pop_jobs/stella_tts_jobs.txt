# MSY Liaison: coordinates grief cube sharing between the local magical girls and the
# planetary authorities.
tts_msy_liaison = {
	category = specialist

	swappable_data = {
		default = {
			condition_string = SPECIALIST_JOB_TRIGGER
			building_icon = building_bureaucratic_1
			icon = bureaucrat
		}
	}

	possible_pre_triggers = {
		has_owner = yes
		is_enslaved = no
		is_being_purged = no
		is_being_assimilated = no
		is_sapient = yes
	}

	possible_precalc = can_fill_specialist_job
	possible = {
		hidden_trigger = { exists = owner }
		owner = { is_gestalt = no }
	}

	resources = {
		category = planet_jobs
		produces = {
			unity = 3
			grief_cube = 1
		}
		upkeep = {
			consumer_goods = 2
		}
	}

	planet_modifier = {
		planet_stability_add = 1
	}

	weight = {
		weight = @specialist_job_weight
	}
}
