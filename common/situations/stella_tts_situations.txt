# To the Stars content: situations
# The Grief Cube Crisis: when cubes run short, soul gems darken, hunts fail and demons spread.
# Started automatically by the game on a grief cube deficit (see deficit_situation in
# common/strategic_resources/stella_strategic_resources.txt). Modelled on the vanilla deficit
# situations. Inspired by Hieronym's "To the Stars". See docs/TTS_CONTENT.md.

situation_tts_grief_cube_deficit = {
	picture = GFX_evt_psionic_battle
	category = negative
	fail_icon = GFX_situation_outcome_meh
	fail_icon_frame = GFX_situation_outcome_frame_green
	complete_icon = GFX_situation_outcome_bankrupcy
	complete_icon_frame = GFX_situation_outcome_frame_red
	situation_log_category = urgent_matters

	on_start = {
		set_situation_flag = deficit_situation
	}

	initial_progress = 15
	stages = {
		deficit_first = {
			icon = GFX_situation_stage_1
			icon_background = GFX_situation_stage_frame_red
			end = 25
			modifier = {
				army_damage_mult = -0.05
				pop_happiness = -0.02
			}
		}
		deficit_second = {
			icon = GFX_situation_stage_2
			icon_background = GFX_situation_stage_frame_red
			end = 50
			modifier = {
				army_damage_mult = -0.10
				pop_happiness = -0.04
				planet_crime_add = 5
			}
		}
		deficit_third = {
			icon = GFX_situation_stage_3
			icon_background = GFX_situation_stage_frame_red
			end = 75
			modifier = {
				army_damage_mult = -0.20
				pop_happiness = -0.06
				planet_crime_add = 10
			}
			on_first_enter = {
				# Girls on the edge of despair: the first demons slip through.
				owner = { country_event = { id = stella_tts_wish.21 } }
			}
		}
		deficit_fourth = {
			icon = GFX_situation_stage_4
			icon_background = GFX_situation_stage_frame_red
			end = 100
			modifier = {
				army_damage_mult = -0.30
				pop_happiness = -0.08
				planet_crime_add = 15
			}
			on_first_enter = {
				owner = { country_event = { id = stella_tts_wish.21 } }
			}
		}
	}

	on_progress_complete = {
		custom_tooltip = situation_tts_grief_cube_deficit_complete_tooltip
		hidden_effect = {
			owner = {
				add_modifier = { modifier = stella_tts_mass_despair days = 3600 }
				country_event = { id = stella_tts_wish.21 }
				country_event = { id = stella_tts_wish.22 }
			}
			destroy_situation = this
		}
	}
	on_fail = {
		custom_tooltip = deficit_situation_finish_tooltip
		hidden_effect = { destroy_situation = this }
	}

	monthly_progress = {
		base = 0
		modifier = {
			add = value:deficit_situation_progress_value|RESOURCE|grief_cube|
			desc = string_expenditures_greater_than_income_grief_cube
			owner = { has_actual_deficit = { RESOURCE = grief_cube } }
		}
		modifier = {
			subtract = 5
			desc = string_not_in_deficit_grief_cube
			owner = {
				resource_income_compare = {
					resource = grief_cube
					value >= 0
				}
			}
		}
		modifier = {
			subtract = 1
			desc = string_resource_reserves_grief_cube
			owner = {
				has_negative_income_with_stockpile = { RESOURCE = grief_cube }
			}
		}
		modifier = {
			subtract = 2
			desc = tts_gc_church_of_hope
			owner = { tts_has_church_of_hope = yes }
		}
		modifier = {
			desc = civilian_difficulty
			mult = 0.1
			situation_monthly_progress > 0
			is_difficulty = 0
			owner = { is_ai = no }
		}
		modifier = {
			desc = cadet_difficulty
			mult = 0.5
			situation_monthly_progress > 0
			is_difficulty = 1
			owner = { is_ai = no }
		}
	}

	approach = {
		name = deficit_approach_do_nothing
		icon = GFX_situation_approach_this_is_fine
		icon_background = GFX_situation_approach_bg_yellow
		default = yes
		on_select = {
			custom_tooltip = deficit_approach_do_nothing_tooltip
		}
		ai_weight = {
			base = 1
		}
	}
	# Strict rationing: the cubes go further, the girls fight weaker.
	approach = {
		name = tts_gc_approach_ration
		icon = GFX_situation_approach_balance_sheet
		icon_background = GFX_situation_approach_bg_yellow
		modifier = {
			armies_grief_cube_upkeep_mult = -0.40
			army_damage_mult = -0.10
		}
		ai_weight = {
			base = 0
			modifier = {
				add = 2
				current_stage = deficit_second
			}
		}
	}
	# Emergency hunts: every available girl out after demons, whatever it costs.
	approach = {
		name = tts_gc_approach_hunts
		icon = GFX_situation_approach_energy_credits
		icon_background = GFX_situation_approach_bg_green
		modifier = {
			country_grief_cube_produces_mult = 0.30
			pop_happiness = -0.05
			army_morale = -0.10
		}
		ai_weight = {
			base = 0
			modifier = {
				add = 3
				current_stage = deficit_third
			}
			modifier = {
				add = 3
				current_stage = deficit_fourth
			}
		}
	}
	# Deal with the Incubators: they always want something in return.
	approach = {
		name = tts_gc_approach_incubators
		icon = GFX_situation_approach_balance_sheet
		icon_background = GFX_situation_approach_bg_green
		modifier = {
			country_grief_cube_produces_mult = 0.20
			country_society_research_produces_mult = -0.10
		}
		ai_weight = {
			base = 0
			modifier = {
				add = 1
				current_stage = deficit_third
			}
		}
	}
}
