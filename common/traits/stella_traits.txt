##Stella Magica Traits
# PORT (4.6): leader classes were merged in 3.x -
#   admiral, general -> commander; governor, ruler -> official; scientist unchanged.
# Bonuses now go through the scope-specific blocks vanilla uses:
#   fleet_modifier (commanders' fleets), army_modifier (commanders' armies),
#   planet_modifier (governed colony), councilor_modifier (empire, while on the council),
#   modifier (the ship a scientist leads).
# Removed deprecated fields: modification, leader_trait, ai_categories,
# potential_crossbreeding_chance. Icons use the Galactic Paragons layered format
# with sprites from interface/stella_traits.gfx.

#General species Puella Magi Trait
trait_magi = {
	cost = 0
	initial = no
	randomized = no
	icon = "gfx/interface/icons/traits/magi.dds"
	improves_leaders = yes
	immortal_leaders = yes
	allowed_archetypes = { BIOLOGICAL }

	# PORT: replaces `modification = no` - cannot be added or removed by genetic modification
	species_possible_add = { always = no }
	species_possible_remove = { always = no }

	modifier = {
		species_leader_exp_gain = 0.25
		leader_skill_levels = 2
	}

	ai_weight = {
		weight = 100
	}
}

# (was: ruler) 2.2 ruler traits affected the whole empire while the leader ruled. In 4.6 a
# leader of any class can be ruler (the events give this trait to commanders, scientists
# and officials), so it is allowed for all classes and only applies while ruler.
leader_trait_ruler_magi = {
	inline_script = {
		script = trait/icon
		CLASS = leader
		ICON = "GFX_leader_trait_stella_magi"
		RARITY = paragon
		COUNCIL = yes
		TIER = none
	}
	immortal_leaders = yes
	randomized = no
	triggered_councilor_modifier = {
		potential = { is_ruler = yes }
		country_unity_produces_mult = 0.05
		country_base_influence_produces_add = 1
	}
	leader_class = { commander official scientist }
	ai_weight = 100
}
