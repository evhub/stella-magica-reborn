#Actually spawn it
country_event = {

	id = stella_apop.5
	is_triggered_only = yes
	hide_window = yes
	
	immediate = {
	random_system = {
		limit = { has_star_flag = stella_oblivion_system }
		create_country = {
				name = "The Fervent"
				type = faction
				flag = {
					icon= {
						category = "special"
						file = "extradimensional_01.dds"
					}
					background= {
						category = "backgrounds"
						file = "circle.dds"
					}
					colors={
						"black"
						"red"
						"null"
						"null"
					}
				}
			}
			
			last_created_country = {
				save_event_target_as = stella_invaders_country
				add_modifier = { modifier = stella_invaders_defense_buff days = -1}
				create_fleet = {
				name = "Extradimensional Invaders"
				settings = { spawn_debris = no }
				effect = {
					set_owner = event_target:stella_invaders_country
					set_fleet_flag = "stella_invaders_fleet"
					
					while = {
						count = 8
						create_ship = {
							name = "Extradimensional Battleship"
							design = "Fervent_Archon"
							graphical_culture = "extra_dimensional_01"
						}
					}
					
					while = {
						count = 14
						create_ship = {
							name = "Extradimensional Cruiser"
							design = "Fervent_Magnus"
							graphical_culture = "extra_dimensional_01"
						}
					}
					
					set_location = {
									target = event_target:stella_oblivion_star
									distance = 150
					}
					
					set_fleet_stance = aggressive
					set_aggro_range = 140
					set_aggro_range_measure_from = self
				}
			}
			}
			
			
			
		}
	}
}
