#Spawn Fervent portal in a system not owned by the player or fallen empire
country_event = {
	id = stella_crisis.8
	hide_window = yes
	is_triggered_only = yes
	
	trigger = {
		NOT = {has_global_flag = fervent_already_spawned}
	}
	immediate = {
		set_global_flag = fervent_already_spawned
		random_system = {
			limit = {
				NOT = {
					any_country = {
						OR = {
							is_country_type = fallen_empire
							is_country_type = awakened_fallen_empire
							is_ai = no
						}
						any_system_within_border = {
							is_same_value = prev
						}
					}
				}
			}
			
			set_star_flag = fervent_origin_system
			save_global_event_target_as = fervent_system
			
			random_system_planet = {
				create_species = {
					name = "Fervent"
					class = EXD
					portrait = exd1
					traits = random
					effect = { save_event_target_as = fervent_species }
				}
				create_country = {
					name = "The Fervent"
					type = "stella_fervent"
					species = event_target:fervent_species
					name_list = "Extradimensional"
					effect = { 
						set_country_flag = unbidden
						add_modifier = {modifier = stella_fervent_modifier days = -1}
						country_event = {id = stella_crisis.900 days = 3600}
						country_event = {id = stella_crisis.902 days = 7200}
						
					}
					flag = {
						icon= {
							category = "special"
							file = "extradimensional_03.dds"
						}
						background= {
							category = "backgrounds"
							file = "circle.dds"
						}
						colors={
							"red"
							"black"
							"null"
							"null"
						}
					}
					
					effect = {
						create_ship_design = { design = "NAME_Void_Shaper" }
						add_ship_design = last_created_design
						create_ship_design = { design = "Fervent_Station" }
						add_ship_design = last_created_design
						save_global_event_target_as = fervent_country
						#establish_communications_no_message = event_target:portal_holder_1
						#set_faction_hostility = {
						#	target = event_target:portal_holder_1
						#	set_hostile = no
						#}
						set_graphical_culture = extra_dimensional_01

					}
				}
				
				event_target:fervent_country = {
					create_fleet = {
						name = "Fervent Portal"
						effect = {
							set_owner = PREV
							set_fleet_flag = fervent_portal
							save_global_event_target_as = fervent_portal_location
								
							create_ship = {
								name = random
								design = "NAME_Unbidden_Portal"
								graphical_culture = "extra_dimensional_01"
							}
							set_location = {
								target = PREVPREV
								distance = 40
								angle = random
							}
							save_event_target_as = dimensional_portal
							fleet_event = { id = stella_crisis.1000 days = 15 } # Second Fleet Arrives
							fleet_event = { id = stella_crisis.1000 days = 30 } # Third Fleet Arrives
							fleet_event = { id = stella_crisis.1000 days = 60 } # Fourth Fleet Arrives
							fleet_event = { id = stella_crisis.1000 days = 100 } # Fifth Fleet Arrives
							#fleet_event = { id = stella_crisis.1000 days = 200 }  Sixth Fleet Arrives
							fleet_event = { id = stella_crisis.1001 days = 20 } # First Constructor Arrives
							fleet_event = { id = stella_crisis.1001 days = 25 } # Second Constructor Arrives
							fleet_event = { id = stella_crisis.1001 days = 160 } # Third Constructor Arrives
							fleet_event = { id = stella_crisis.1001 days = 310 } # Fourth Constructor Arrives
							event_target:fervent_country = { country_event = { id = stella_crisis.1002 days = 360 } } # Spawn Cycle starts
							
						}
					}
					
					create_fleet = {
						settings = { spawn_debris = no garrison = yes}
						effect = {
							set_owner = PREV

							while = {
								count = 7
								create_ship = {
									name = random
									design = "Fervent_Archon"
									graphical_culture = "extra_dimensional_01"
								}
							}
							while = {
								count = 12
								create_ship = {
									name = random
									design = "Fervent_Magnus"
									graphical_culture = "extra_dimensional_01"
								}
							}
							while = {
								count = 14
								create_ship = {
									name = random
									design = "Fervent_Shadow"
									graphical_culture = "extra_dimensional_01"
								}
							}
							set_location = {
								target = event_target:dimensional_portal
								distance = 5
								angle = random
							}
							set_fleet_stance = aggressive
							set_aggro_range = 500
							set_aggro_range_measure_from = self
						}
					}
				}

			}
			
			if = {
				limit = { exists = starbase }
				starbase = { fleet = { destroy_fleet = this } }
			}
			create_starbase = {
				size = starbase_exd
				owner = event_target:fervent_country
			}
			
			every_playable_country = {
				establish_communications_no_message = event_target:fervent_country
			}
		}
		
	
	}
	
	after = {
		#Let every other country know about the invasion
		every_playable_country = {
			limit = {NOT = {has_country_flag = triggered_fervent_invasion}}
			country_event = {id = stella_crisis.9 days = 3}
		}
	}
}

#The Fervent have a max of 200 ships
country_event = {
	id = stella_crisis.1002
	hide_window = yes
	
	is_triggered_only = yes
	
	trigger = {
			is_country_type = stella_fervent
			count_owned_ships = {
				limit = { is_ship_size = dimensional_portal_ed }
				count > 0
			}
	}
	
	immediate = {
		#Spawn some constructors if there are 2 or less
		country_event = {id = stella_crisis.1005}
		
		#At max ships
		if = {
			limit = {
				num_ships > 200
			}
			country_event = { id = stella_crisis.1002 days = 500 random = 220 } # Check again later
		}
		else = {
			country_event = { id = stella_crisis.1003 days = 300 random = 220 }
		}
		
		
	}
}

#Military reinforcements
country_event = {
	id = stella_crisis.1003
	hide_window = yes
	is_triggered_only = yes
	
	trigger = {
			is_country_type = stella_fervent
			count_owned_ships = {
				limit = { is_ship_size = dimensional_portal_ed }
				count > 0
			}
			count_owned_ships = {
				limit = { is_ship_size = construction_ship_ed }
				count < 3
			}
		
	}
	
	immediate = {
		country_event = { id = stella_crisis.1002 }
		if = {
			limit = { is_country_type = stella_fervent }
			while = {
