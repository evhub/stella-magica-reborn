	desc = "stella_dim.4.desc"
	is_triggered_only = yes

	immediate = {
		
		capital_scope = { save_event_target_as = capital_system } 
			random_system = {
				limit = {
					distance = {
						type = hyperlane
						source = event_target:capital_system
						use_bypasses = no
						min_jumps = 3
						max_jumps = 4
					}
					
					OR = {
					
						is_inside_border = ROOT
						
						NAND = {
							exists = space_owner
							has_star_flag = hostile_system
						}
					}	
				}
				spawn_system = {
					min_distance >= 10
					max_distance <= 10
					initializer = "epitaph_system_initializer"
				}
			}
			random_system = {
				limit = {
					has_star_flag = stella_tomb_system
				}
				set_star_flag = sdim_system_1
				save_event_target_as = sdim_system_1
			}

				id = between_dims_poi.1
				name = "between_dims_1_poi"
				desc = "between_dims_1_poi_desc"
				event_chain = "between_dims_chain"
				location = event_target:sdim_system_1

		name = OK
	}
