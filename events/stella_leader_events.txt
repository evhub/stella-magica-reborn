namespace = stella_leader

#Gives leaders the right traits when they change jobs (Become ruler, etc.)
#Check on action for triggers
# PORT: originally fired from on_ruler_back_to_pre_ruler_class, which no longer exists in 4.6 (rulers keep their class).
# Scopes expected: This = country, From = leader. on_ruler_removed (From = previous ruler) and on_leader_spawned
# (From = new leader) both provide this.
# 4.6 classes: admiral + general -> commander, governor (+ ruler) -> official, scientist unchanged.
# Commanders get leader_trait_admiral_magi (the first branch of the original); leader_trait_general_magi is still
# counted as "already has a magi trait".
country_event = {
	id = stella_leader.1
	is_triggered_only = yes
	hide_window = yes

	trigger = {
		from = {
			species = { has_trait = trait_magi }
			NOR = {
				has_trait = leader_trait_admiral_magi
				has_trait = leader_trait_general_magi
				has_trait = leader_trait_governor_magi
				has_trait = leader_trait_scientist_magi
			}
		}
	}

	immediate = {
		from = {
			if = {
				limit = { leader_class = commander }
				add_trait = { trait = leader_trait_admiral_magi show_message = no }
			}
			else_if = {
				limit = { leader_class = official }
				add_trait = { trait = leader_trait_governor_magi show_message = no }
			}
			else_if = {
				limit = { leader_class = scientist }
				add_trait = { trait = leader_trait_scientist_magi show_message = no }
			}
		}
	}
}

#Detects if any of the special leaders died or not
country_event = {
	id = stella_leader.2
	is_triggered_only = yes
	hide_window = yes
	
	trigger = {
		from = {
			species = { has_trait = trait_magi }
		}
	}
	
	# PORT: the original ran these effects inside from = { } (leader scope), but remove_country_flag and country_event
	# are country-only in 4.6, so the flags were never cleared. The leader is now checked from country scope (This = country).
	immediate = {
		if = {
			limit = { from = { has_leader_flag = stella_mami } }
			remove_country_flag = stella_mami_alive
		}
		else_if = {
			limit = { from = { has_leader_flag = stella_kyoko } }
			remove_country_flag = stella_kyoko_alive
		}
		else_if = {
			limit = { from = { has_leader_flag = stella_yuma } }
			remove_country_flag = stella_yuma_alive
		}
		else_if = {
			limit = { from = { has_leader_flag = stella_homura } }
			remove_country_flag = stella_homura_alive
		}
		else_if = {
			limit = { from = { has_leader_flag = stella_sayaka } }
			remove_country_flag = stella_sayaka_alive
		}
		else_if = {
			limit = { from = { has_leader_flag = stella_madoka } }
			remove_country_flag = stella_madoka_alive
			country_event = {id = stella_leader.10 days = 10}
		}
	}
}

#Leader spawner events
#stella_leader.3 = Mami
#stella_leader.4 = Kyoko
#stella_leader.5 = Yuma
#stella_leader.6 = Homura
#stella_leader.7 = Sayaka
#stella_leader.8 = Madoka
#stella_leader.9 = Ultra Instinct Madoka
#
# PORT (4.6): admiral -> commander, governor -> official. The create_leader trait list keeps the original "trait = x"
# form (level 0 = always added, no trait selection consumed).
# add_ruler_trait no longer exists (4.x has no separate ruler trait slot), so leader_trait_ruler_magi is now added
# as a normal trait. For this to work on commanders, scientists and officials, the trait needs leader_class = all, and
# to keep the 2.2 behaviour its bonuses should only apply while the leader is ruler (e.g. gated on is_ruler = yes).

country_event = {

	id = stella_leader.3
	is_triggered_only = yes
	hide_window = yes
	
	trigger = {
		NOT = { has_country_flag = stella_mami_alive }
	}
	
	immediate = {
		set_country_flag = stella_mami_alive
		create_species = {
			name = "Puella Magi"
			class = PuellaMagi
			portrait = mami
			traits = {
				trait = trait_pc_continental_preference 
				trait = trait_charismatic
				trait = trait_magi
			}
		}
		
		create_leader = {
			gender = female
			class = commander # PORT: admiral -> commander
			species = last_created
			name = "Mami Tomoe"
			skill = 3
			randomize_traits = no # PORT: 4.6 rolls extra random traits by default; keep only the scripted ones as in the original
			traits = {
				trait = leader_trait_admiral_magi
				trait = leader_trait_trio_finish
			}
			
		}
		
		
		last_created_leader  = {
			set_leader_flag  = stella_mami
			add_trait = { trait = leader_trait_ruler_magi show_message = no } # PORT: add_ruler_trait removed in 4.6, see note above stella_leader.3

		}
	}
}
