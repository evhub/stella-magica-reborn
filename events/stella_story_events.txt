#Spawn traced location
country_event = {
	id = stella_story.4
	title = "stella_story.4.name"
	desc = "stella_story.4.desc"
	picture = GFX_evt_star_chart
	show_sound = event_sensor_ping
	
	is_triggered_only = yes
	
	immediate = {
		capital_scope = {  save_global_event_target_as = capital_system } 
		hidden_effect = {
			random_system = {
				limit = {
					distance = {
						type = hyperlane
						source = event_target:capital_system
						use_bypasses = no
						min_jumps = 1
						max_jumps = 3
					}
					
					OR = {
					
						is_inside_border = ROOT
						
						NAND = {
							exists = space_owner
							has_star_flag = hostile_system
						}
					}
				}
				save_event_target_as = homu_system_1
				set_star_flag = homu_system_1
			}
			
			create_point_of_interest = {
				id = chasing_shadows_poi.1
				name = "chasing_shadows_1_poi"
				desc = "chasing_shadows_1_poi_desc"
				event_chain = "chasing_shadows_chain"
				location = event_target:homu_system_1
			}
		}
		
		
	}
	option = {
		name = stella_story.4.op1
	}
}

#Ship enters the first marked system (Hidden event)
ship_event = {
	id = stella_story.5
	hide_window = yes
	is_triggered_only = yes
	
	trigger = {
		owner = {
			is_ai = no
			has_event_chain = "chasing_shadows_chain"
			NOT = { has_country_flag = homu_system_1_entered }
		}
		solar_system = { has_star_flag = homu_system_1 }
	}
	
	
	immediate = {
			owner = { set_country_flag = homu_system_1_entered }
			owner = { remove_point_of_interest = chasing_shadows_poi.1}
			ship_event = { id = stella_story.6 }
	}
}
