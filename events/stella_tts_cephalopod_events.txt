# To the Stars content: the Cephalopod War
# The Cephalopods are a real AI empire, the /Thinker-preserving Tentacle (preset
# tts_thinker_preserving, always spawned). Their mandate from the Thinker Consensus is to stop
# civilizations using "reality-distortion technology", so from the mid-game they send enforcement
# task forces against magic-using empires (Magical Legacy civic). A mothership anchors a forward
# position and raiding swarms sortie from it until it is destroyed; the wreck can then be studied.
# Inspired by Hieronym's "To the Stars". See docs/TTS_CONTENT.md.

namespace = stella_tts_ceph

# Game start (Thinker empire): register the empire, its designs and its technological head start.
country_event = {
	id = stella_tts_ceph.40
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = tts_thinker_preserving
	}

	immediate = {
		save_global_event_target_as = tts_thinker_empire
		give_technology = { tech = tech_lasers_2 message = no }
		give_technology = { tech = tech_shields_2 message = no }
		give_technology = { tech = tech_strike_craft_1 message = no }
		give_technology = { tech = tech_destroyers message = no }
		create_ship_design = { design = "NAME_tts_Ceph_Skirmisher" }
		add_ship_design = last_created_design
		create_ship_design = { design = "NAME_tts_Ceph_Harrier" }
		add_ship_design = last_created_design
		create_ship_design = { design = "NAME_tts_Ceph_Leviathan" }
		add_ship_design = last_created_design
		create_ship_design = { design = "NAME_tts_Ceph_Mothership" }
		add_ship_design = last_created_design
	}
}

# Yearly check (Thinker empire): from year 60, a small chance each year to send an enforcement
# task force against a magic-using empire. One campaign at a time, with a long pause between them.
country_event = {
	id = stella_tts_ceph.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = tts_thinker_preserving
		years_passed >= 60
		exists = capital_scope
		NOR = {
			has_global_flag = stella_tts_ceph_war
			has_global_flag = stella_tts_ceph_cooldown
			has_global_flag = stella_tts_ceph_disabled
		}
		any_playable_country = {
			has_valid_civic = civic_magical_legacy
			exists = capital_scope
		}
	}

	immediate = {
		random_list = {
			90 = { }
			10 = { country_event = { id = stella_tts_ceph.2 } }
		}
	}
}

# Launch an enforcement task force (Thinker empire scope).
country_event = {
	id = stella_tts_ceph.2
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		NOT = { has_global_flag = stella_tts_ceph_war }
	}

	immediate = {
		random_playable_country = {
			limit = {
				has_valid_civic = civic_magical_legacy
				exists = capital_scope
			}
			save_global_event_target_as = tts_ceph_first_target
		}
		if = {
			limit = { exists = event_target:tts_ceph_first_target }
			random_system = {
				limit = {
					NOT = { exists = owner }
					distance = {
						source = event_target:tts_ceph_first_target.capital_scope.solar_system
						use_bypasses = no
						min_jumps = 6
						max_jumps = 14
					}
				}
				set_global_flag = stella_tts_ceph_war
				set_star_flag = stella_tts_ceph_base
				save_global_event_target_as = tts_ceph_base_system
			}
		}
		if = {
			limit = { has_global_flag = stella_tts_ceph_war }
			# The mothership and its guard hold the forward position.
			create_fleet = {
				name = "NAME_tts_Ceph_Mothership_Fleet"
				settings = { spawn_debris = yes }
				effect = {
					set_owner = root
					set_fleet_flag = stella_tts_ceph_mothership
					create_ship = {
						name = random
						design = "NAME_tts_Ceph_Mothership"
						graphical_culture = "molluscoid_01"
					}
					while = {
						count = 2
						create_ship = {
							name = random
							design = "NAME_tts_Ceph_Leviathan"
							graphical_culture = "molluscoid_01"
						}
					}
					set_location = {
						target = event_target:tts_ceph_base_system
						distance = 20
						angle = random
					}
					set_aggro_range = 50
				}
			}
			# Their blink drives are no longer a secret.
			if = {
				limit = { NOT = { has_technology = tech_jump_drive_1 } }
				give_technology = { tech = tech_jump_drive_1 message = no }
			}
			while = {
				count = 2
				country_event = { id = stella_tts_ceph.11 }
			}
			country_event = { id = stella_tts_ceph.10 days = 720 }
			event_target:tts_ceph_first_target = {
				country_event = { id = stella_tts_ceph.3 days = 10 }
			}
		}
	}
}

# The task force arrives (target empire).
country_event = {
	id = stella_tts_ceph.3
	title = stella_tts_ceph.3.name
	desc = stella_tts_ceph.3.desc
	picture = GFX_evt_unknown_ships
	is_triggered_only = yes

	option = {
		name = stella_tts_ceph.3.a
	}
	option = {
		name = stella_tts_ceph.3.b
		trigger = { has_valid_civic = civic_tts_msy }
		add_resource = { grief_cube = 50 }
	}
}

# Reinforcement cycle (Thinker empire scope).
country_event = {
	id = stella_tts_ceph.10
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = { any_owned_fleet = { has_fleet_flag = stella_tts_ceph_mothership } }
			if = {
				limit = {
					count_owned_fleet = {
						limit = { has_fleet_flag = stella_tts_ceph_raiders }
						count < 6
					}
				}
				country_event = { id = stella_tts_ceph.11 }
			}
			country_event = { id = stella_tts_ceph.10 days = 720 }
		}
		else = {
			# The mothership is gone: this campaign is over.
			remove_global_flag = stella_tts_ceph_war
			set_global_flag = stella_tts_ceph_won
			set_timed_global_flag = { flag = stella_tts_ceph_cooldown days = 7200 }
			if = {
				limit = { exists = event_target:tts_ceph_first_target }
				event_target:tts_ceph_first_target = {
					country_event = { id = stella_tts_ceph.20 }
				}
			}
		}
	}
}

# Send one raiding swarm from the forward position (Thinker empire scope).
country_event = {
	id = stella_tts_ceph.11
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		exists = event_target:tts_ceph_base_system
	}

	immediate = {
		create_fleet = {
			name = "NAME_tts_Ceph_Raid_Fleet"
			effect = {
				set_owner = root
				set_fleet_flag = stella_tts_ceph_raiders
				while = {
					count = 4
					create_ship = {
						name = random
						design = "NAME_tts_Ceph_Skirmisher"
						graphical_culture = "molluscoid_01"
					}
				}
				while = {
					count = 2
					create_ship = {
						name = random
						design = "NAME_tts_Ceph_Harrier"
						graphical_culture = "molluscoid_01"
					}
				}
				set_location = {
					target = event_target:tts_ceph_base_system
					distance = 30
					angle = random
				}
				set_fleet_stance = aggressive
			}
		}
	}
}
