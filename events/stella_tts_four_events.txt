# To the Stars content: the Mitakihara Four and the MSY's divisions
# Tomoe Mami, Sakura Kyouko, Chitose Yuma and Akemi Homura were the nucleus around which the MSY
# grew. Mami became Field Marshal, Kyouko founded the Cult of Hope (see stella_tts_hope.3), Yuma
# leads the MSY, and Kuroi Kana directs its Black Heart. Inspired by Hieronym's "To the Stars".
# The text is original. See docs/TTS_CONTENT.md.

namespace = stella_tts_four

# Yearly roll (once MagOps exists): Tomoe Mami takes command.
country_event = {
	id = stella_tts_four.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_valid_civic = civic_magical_legacy
		has_country_flag = stella_tts_magops_founded
		NOT = { has_country_flag = stella_tts_mami_joined }
		exists = capital_scope
	}

	immediate = {
		random_list = {
			90 = { }
			10 = { country_event = { id = stella_tts_four.2 } }
		}
	}
}

# The Field Marshal.
country_event = {
	id = stella_tts_four.2
	title = stella_tts_four.2.name
	desc = stella_tts_four.2.desc
	picture = GFX_evt_fleet_neutral
	is_triggered_only = yes

	immediate = {
		set_country_flag = stella_tts_mami_joined
	}

	option = {
		name = stella_tts_four.2.a
		create_leader = {
			class = commander
			species = owner_main_species
			name = "NAME_tts_mami"
			gender = female
			skill = 4
			randomize_traits = no
		}
		last_created_leader = {
			set_leader_flag = stella_tts_mami
			set_leader_flag = stella_tts_magical_girl
			set_immortal = yes	# magical girls do not age
			add_trait = { trait = leader_trait_tts_field_marshal show_message = no }
			add_trait = { trait = leader_trait_tts_taccomp show_message = no }
		}
	}
}
