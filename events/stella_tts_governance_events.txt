# To the Stars content: Governance and society
# Governance's Directorate decides through committees of Representatives (Military Affairs,
# Colonial Affairs, the Science Division, Public Order); the Mages First movement once argued that
# magical girls should come first; and Version Two TacComps were never supposed to develop
# personalities. Inspired by Hieronym's "To the Stars". The text is original. See docs/TTS_CONTENT.md.

namespace = stella_tts_gov

# Yearly roll (Governance): roughly once a decade the Directorate reviews its priorities.
country_event = {
	id = stella_tts_gov.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		tts_has_governance = yes
		years_passed >= 5
		NOT = { has_country_flag = stella_tts_gov_recent }
	}

	immediate = {
		random_list = {
			85 = { }
			15 = {
				set_timed_country_flag = { flag = stella_tts_gov_recent days = 3600 }
				country_event = { id = stella_tts_gov.2 }
			}
		}
	}
}
