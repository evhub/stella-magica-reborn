# Yearly roll (Black Heart director on the council, once the Cephalopods have attacked).
country_event = {
	id = stella_tts_hidden.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = stella_tts_kana_joined
		OR = {
			has_global_flag = stella_tts_ceph_war
			has_global_flag = stella_tts_ceph_won
		}
		NOT = { has_country_flag = stella_tts_hidden_started }
	}

	immediate = {
		random_list = {
			80 = { }
			20 = {
				set_country_flag = stella_tts_hidden_started
				country_event = { id = stella_tts_hidden.2 }
			}
		}
	}
}
