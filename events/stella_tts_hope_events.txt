# To the Stars content: the Cult of Hope
# Inspired by Hieronym's "To the Stars". See docs/TTS_CONTENT.md.

namespace = stella_tts_hope

# Yearly roll (empires that recognise the Church of Hope): roughly one vision every eight years.
country_event = {
	id = stella_tts_hope.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		tts_has_church_of_hope = yes
		exists = capital_scope
	}

	immediate = {
		random_list = {
			88 = { }
			12 = {
				# Visions come twice as often to an empire with a Ribbon Chamber.
				modifier = {
					factor = 2
					any_owned_planet = { has_building = building_tts_ribbon_chamber }
				}
				country_event = { id = stella_tts_hope.2 }
			}
		}
	}
}

# A vision of the Goddess.
country_event = {
	id = stella_tts_hope.2
	title = stella_tts_hope.2.name
	desc = stella_tts_hope.2.desc
	picture = GFX_evt_temple_of_whispers
	is_triggered_only = yes

	option = {
		name = stella_tts_hope.2.a
		capital_scope = {
			add_modifier = { modifier = stella_tts_miracle days = 1800 }
		}
		add_resource = { unity = 150 }
		ai_chance = { factor = 2 }
	}
	option = {
		name = stella_tts_hope.2.b
		add_resource = { society_research = 250 }
		ai_chance = { factor = 1 }
	}
}

# The Church of Hope is organised (tech researched): its first priestess joins the council.
country_event = {
	id = stella_tts_hope.3
	title = stella_tts_hope.3.name
	desc = stella_tts_hope.3.desc
	picture = GFX_evt_temple_of_whispers
	is_triggered_only = yes

	trigger = {
		last_increased_tech = tech_tts_theology_of_hope
		NOT = { has_country_flag = stella_tts_church_founded }
	}

	immediate = {
		set_country_flag = stella_tts_church_founded
	}

	option = {
		name = stella_tts_hope.3.a
		create_leader = {
			class = official
			species = owner_main_species
			name = "NAME_tts_kyouko"
			gender = female
			skill = 3
			randomize_traits = no
		}
		last_created_leader = {
			add_trait = { trait = leader_trait_tts_priestess_of_hope show_message = no }
		}
	}
}

# Yearly roll (Magical Legacy empires): the Church of Hope asks for recognition.
# In the fic it is a magical girls' faith inside a secular society, so any ethics can take it.
country_event = {
	id = stella_tts_hope.9
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_valid_civic = civic_magical_legacy
		years_passed >= 3
		exists = capital_scope
		NOT = { has_country_flag = stella_tts_church_decided }
	}

	immediate = {
		random_list = {
			75 = { }
			25 = { country_event = { id = stella_tts_hope.10 } }
		}
	}
}
