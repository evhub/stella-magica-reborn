# To the Stars content: the Incubators
# In the fic the Incubators cooperate with humanity under a wary treaty: they offer contracts,
# collect what they want, take a puzzling interest in the Church's visions, and seem to know more
# about the Cephalopods than they say. Kyubey speaks for them. Inspired by Hieronym's
# "To the Stars". The text is original. See docs/TTS_CONTENT.md.

namespace = stella_tts_qb

# Yearly roll (Magical Legacy empires): Kyubey comes calling, every decade or so.
country_event = {
	id = stella_tts_qb.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_valid_civic = civic_magical_legacy
		years_passed >= 10
		exists = capital_scope
		NOT = { has_country_flag = stella_tts_qb_recent }
	}

	immediate = {
		random_list = {
			85 = { }
			10 = {
				set_timed_country_flag = { flag = stella_tts_qb_recent days = 3600 }
				country_event = { id = stella_tts_qb.2 }
			}
			5 = {
				modifier = { factor = 0 has_resource = { type = grief_cube amount < 60 } }
				set_timed_country_flag = { flag = stella_tts_qb_recent days = 3600 }
				country_event = { id = stella_tts_qb.3 }
			}
		}
	}
}
