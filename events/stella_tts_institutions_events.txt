# The MSY charter (game start, MSY civic)

		set_country_flag = stella_tts_msy_charter
	}

	option = {
		name = stella_tts_inst.1.a
		add_resource = { grief_cube = 100 }
	}
}

# TacComp rollout (tech researched): equip the best commander first
country_event = {
	id = stella_tts_inst.3
	title = stella_tts_inst.3.name
	desc = stella_tts_inst.3.desc
	picture = GFX_evt_robot_assembly_plant
	is_triggered_only = yes

	trigger = {
		last_increased_tech = tech_tts_taccomp
		any_owned_leader = {
			leader_class = commander
			NOT = { has_trait = leader_trait_tts_taccomp }
		}
	}

	option = {
		name = stella_tts_inst.3.a
		random_owned_leader = {
			limit = {
				leader_class = commander
				NOT = { has_trait = leader_trait_tts_taccomp }
			}
			add_trait = {
				trait = leader_trait_tts_taccomp
				show_message = no
			}
		}
	}
}

# MagOps founded (tech researched): a teleporter volunteers to lead the first squad
country_event = {
	id = stella_tts_inst.4
	title = stella_tts_inst.4.name
	desc = stella_tts_inst.4.desc
	picture = GFX_evt_ground_combat
	is_triggered_only = yes

	trigger = {
		last_increased_tech = tech_tts_magops_doctrine
		has_valid_civic = civic_magical_legacy
		NOT = { has_country_flag = stella_tts_magops_founded }
	}

	immediate = {
		set_country_flag = stella_tts_magops_founded
	}

	option = {
		name = stella_tts_inst.4.a
		create_leader = {
			class = commander
			species = owner_main_species
			name = random
			skill = 3
			randomize_traits = no
		}
		last_created_leader = {
			add_trait = {
				trait = leader_trait_tts_teleporter
				show_message = no
			}
			set_leader_flag = stella_tts_magops_founder
		}
	}
}

# The Unified Earth origin starts with cortical implants already in everyday use.
